CREATE TABLE sponsor (
    id_sponsor NUMBER PRIMARY KEY,
    nume VARCHAR2(255) NOT NULL,
    email VARCHAR2(255) NOT NULL UNIQUE
);

CREATE TABLE reclama (
    id_reclama NUMBER PRIMARY KEY,
    id_sponsor NUMBER,

    CONSTRAINT fk_reclama_sponsor FOREIGN KEY (id_sponsor) REFERENCES sponsor(id_sponsor)
);

CREATE TABLE jurat (
    id_jurat NUMBER PRIMARY KEY,
    nume VARCHAR2(255) NOT NULL,
    telefon VARCHAR2(20) NOT NULL UNIQUE,
    email VARCHAR2(255) NOT NULL UNIQUE,
    salariu NUMBER
);

CREATE TABLE stapan (
    id_stapan NUMBER PRIMARY KEY,
    nume VARCHAR2(255) NOT NULL,
    telefon VARCHAR2(20) NOT NULL UNIQUE
);

CREATE TABLE catel (
    id_caine NUMBER PRIMARY KEY,
    nume VARCHAR2(255) NOT NULL,
    rasa VARCHAR2(255),
    id_stapan NUMBER,

    CONSTRAINT fk_catel_stapan FOREIGN KEY (id_stapan) REFERENCES stapan(id_stapan)
);

CREATE TABLE premiu (
    id_premiu NUMBER PRIMARY KEY,
    tip_premiu VARCHAR2(20) CHECK (tip_premiu IN ('loc1', 'loc2', 'loc3', 'special', 'eleganta')) NOT NULL,
    suma_de_bani NUMBER,
    id_sponsor NUMBER,

    CONSTRAINT fk_premiu_sponsor FOREIGN KEY (id_sponsor) REFERENCES sponsor(id_sponsor),
    id_jurat NUMBER,
    CONSTRAINT fk_premiu_jurat FOREIGN KEY (id_jurat) REFERENCES jurat(id_jurat),
    id_caine NUMBER,
    CONSTRAINT fk_premiu_caine FOREIGN KEY (id_caine) REFERENCES catel(id_caine)
);

CREATE TABLE proba (
    id_proba NUMBER PRIMARY KEY,
    tip_proba VARCHAR2(20) NOT NULL,
    pret_bilet NUMBER
);

CREATE TABLE participare (
    id_caine NUMBER,
    id_proba NUMBER,

    CONSTRAINT pk_participare PRIMARY KEY (id_caine, id_proba),
    CONSTRAINT fk_participare_caine FOREIGN KEY (id_caine) REFERENCES catel(id_caine),
    CONSTRAINT fk_participare_proba FOREIGN KEY (id_proba) REFERENCES proba(id_proba)
);

CREATE TABLE spectator (
    id_spectator NUMBER PRIMARY KEY,
    nume VARCHAR2(255) NOT NULL
);

CREATE TABLE accesare (
    id_proba NUMBER,
    id_spectator NUMBER,

    CONSTRAINT pk_accesare PRIMARY KEY (id_proba, id_spectator),
    CONSTRAINT fk_accesare_proba FOREIGN KEY (id_proba) REFERENCES proba(id_proba),
    CONSTRAINT fk_accesare_spectator FOREIGN KEY (id_spectator) REFERENCES spectator(id_spectator)
);

-- 6
-- Folosind colecții, să se obțină informații despre câinii fiecărui stăpân și câștigul adus de acești câini 
-- stăpânilor, apoi pe baza colecțiilor construite să se de termine suma de bani medie câștigată de fiecare stăpân.
CREATE OR REPLACE PROCEDURE sase
IS
    TYPE castig_caine IS RECORD(
        id catel.id_caine%TYPE,
        bani NUMBER
    );

    TYPE castiguri_stapan IS VARRAY(100) OF castig_caine;

    TYPE stapani IS TABLE OF castiguri_stapan INDEX BY PLS_INTEGER;
    
    v_castig castig_caine;
    v_stapan castiguri_stapan;
    t_stapani stapani;

    i NUMBER;
    c NUMBER;
    sum NUMBER;
BEGIN
    FOR stap IN (SELECT id_stapan id_s FROM stapan) LOOP
        v_stapan := castiguri_stapan();

        FOR info IN (SELECT c.id_caine id_c, p.suma_de_bani suma
                    FROM catel c
                    INNER JOIN premiu p on p.id_caine = c.id_caine
                    WHERE c.id_stapan = stap.id_s) LOOP
            v_castig := castig_caine();
            v_castig.id := info.id_c;
            v_castig.bani := NVL(info.suma, 0);

            v_stapan.extend(1);
            v_stapan(v_stapan.last) := v_castig;
        END LOOP;

        t_stapani(stap.id_s) := v_stapan;
    END LOOP;

    i := NULL;
    LOOP
        IF i IS NULL THEN
            i := t_stapani.first;
        ELSE
            i := t_stapani.next(i);
        END IF;

        DBMS_OUTPUT.PUT_LINE('Stapanul cu id ' || i || ' are');

        c := 0;
        sum := 0;

        IF t_stapani(i).COUNT > 0 THEN
            FOR j IN t_stapani(i).first..t_stapani(i).last LOOP
                c := c + 1;
                sum := sum + t_stapani(i)(j).bani;
            END LOOP;
        END IF;

        IF c > 0 THEN
            DBMS_OUTPUT.PUT_LINE(c || ' intrari si castig mediu ' || TRUNC(sum / c, 2) || '.');
        ELSE
            DBMS_OUTPUT.PUT_LINE('0 intrari.');
        END IF;

        EXIT WHEN i = t_stapani.last;
    END LOOP;
END;

BEGIN
    sase;
END;

-- 7
-- Pentru fiecare cațel, să se determine probele la care a participat.
CREATE OR REPLACE PROCEDURE sapte
IS
    CURSOR probele(id_c catel.id_caine%type) IS
        SELECT
            p.id_proba id_p,
            p.tip_proba tip_p,
            p.pret_bilet pret
        FROM proba p
        WHERE id_c in (SELECT id_caine FROM participare WHERE id_proba = p.id_proba);

    CURSOR catei IS
        SELECT id_caine id_c
        FROM catel;

    id_c catel.id_caine%TYPE;
BEGIN
    OPEN catei;
    LOOP
        FETCH catei INTO id_c;
        EXIT WHEN catei%NOTFOUND;
        
        DBMS_OUTPUT.PUT_LINE('Pentru catelul cu id ' || id_c || ' avem:');

        FOR proba IN probele(id_c) LOOP
            DBMS_OUTPUT.PUT_LINE('-> proba ' || proba.id_p || ' de tip ' || proba.tip_p || ' la care biletele au costat ' || proba.pret || ';');
        END LOOP;
    END LOOP;
END;

BEGIN
    sapte;
END;

-- 8
-- Să se determine numârul de premii câștigate de cățeii stăpânului cu numele dat ca parametru, 
-- tratând cazul în care se potrivesc mai mulți și cazul în care nu se potrivește niciunul.
CREATE OR REPLACE FUNCTION opt(nume_stapan stapan.nume%TYPE) 
RETURN NUMBER
IS
    id_s stapan.id_stapan%TYPE;
    nr_premii number;
BEGIN
    SELECT id_stapan
    INTO id_s
    FROM stapan
    WHERE UPPER(nume) LIKE '%' || UPPER(nume_stapan) || '%';

    SELECT COUNT(DISTINCT p.id_premiu)
    INTO nr_premii
    FROM stapan s
    INNER JOIN catel c ON c.id_stapan = s.id_stapan
    INNER JOIN premiu p ON p.id_caine = c.id_caine
    WHERE s.id_stapan = id_s;

    RETURN nr_premii;

EXCEPTION
    WHEN TOO_MANY_ROWS THEN
        RAISE_APPLICATION_ERROR(-20008, 'Prea multi stapani cu acelasi nume.');
    WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(-20009, 'Nu avem niciun astfel de stapan.');
END;

-- 9
-- Pentru o rasă și un tip de probă date, să se stabileasca câți spectatori 
-- s-au uitat la probe de acel tip în care a fost prezent un câine de acea rasa.
CREATE OR REPLACE PROCEDURE noua(
    rasa_ceruta IN catel.rasa%TYPE,
    proba_ceruta IN proba.tip_proba%TYPE,
    nr_spectatori OUT NUMBER
)
IS
    RASA_INEXISTENTA EXCEPTION;
    PROBA_INEXISTENTA EXCEPTION;

    nr NUMBER;
BEGIN
    SELECT COUNT(*)
    INTO nr
    FROM catel
    WHERE rasa = rasa_ceruta;

    IF nr = 0 THEN
        RAISE RASA_INEXISTENTA;
    END IF;

    SELECT COUNT(*)
    INTO nr
    FROM proba
    WHERE tip_proba = proba_ceruta;

    IF nr = 0 THEN
        RAISE PROBA_INEXISTENTA;
    END IF;

    SELECT COUNT(DISTINCT s.id_spectator)
    INTO nr
    FROM catel c
    INNER JOIN participare p ON c.id_caine = p.id_caine
    INNER JOIN proba pr ON pr.id_proba = p.id_proba
    INNER JOIN accesare a ON pr.id_proba = a.id_proba
    INNER JOIN spectator s ON a.id_proba = s.id_spectator;

    nr_spectatori := nr;
EXCEPTION
    WHEN RASA_INEXISTENTA THEN
        DBMS_OUTPUT.PUT_LINE('Nu avem catei din aceasta rasa.');
    WHEN PROBA_INEXISTENTA THEN
        DBMS_OUTPUT.PUT_LINE('Nu avem asemenea proba.');
END;

DECLARE
    nr_spectatori NUMBER;
BEGIN
    noua('xoloitzcuintle', 'atitudine', nr_spectatori);
    DBMS_OUTPUT.PUT_LINE('Numarul de spectatori este: ' || nr_spectatori || '.');

    noua('bulldog', 'atitudine', nr_spectatori);
    noua('xoloitzcuintle', 'examen SGBD', nr_spectatori);
END;

CREATE OR REPLACE TRIGGER zece
    BEFORE UPDATE OF suma_de_bani ON premiu
BEGIN
    IF TO_CHAR(SYSDATE, 'D') = '5' THEN
        RAISE_APPLICATION_ERROR(-20010, 'Nu se poate schimba suma de bani astazi.');
    END IF;
END;

    END IF;
END;
